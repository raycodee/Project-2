cmake_minimum_required(VERSION 3.16)
project(GEvents LANGUAGES CXX)

option(GEVENTS_ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)

add_library(gevents src/HashTable.cpp src/Text.cpp src/Filters.cpp src/Ranking.cpp)
target_include_directories(gevents PUBLIC include)
target_compile_features(gevents PUBLIC cxx_std_17)
set_target_properties(gevents PROPERTIES CXX_EXTENSIONS OFF)

if(GEVENTS_ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(gevents PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(gevents PUBLIC -fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "GEVENTS_ENABLE_SANITIZERS requires Clang or GCC")
    endif()
endif()

add_executable(raymond_demo examples/raymond_demo.cpp)
target_link_libraries(raymond_demo PRIVATE gevents)

include(CTest)
set(gevents_targets gevents raymond_demo)
if(BUILD_TESTING)
    foreach(test_name hash_table filter ranking)
        add_executable(${test_name}_tests tests/${test_name}_tests.cpp)
        target_link_libraries(${test_name}_tests PRIVATE gevents)
        add_test(NAME ${test_name} COMMAND ${test_name}_tests)
        list(APPEND gevents_targets ${test_name}_tests)
    endforeach()
endif()

foreach(target_name IN LISTS gevents_targets)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    elseif(MSVC)
        target_compile_options(${target_name} PRIVATE /W4 /WX)
    endif()
endforeach()
