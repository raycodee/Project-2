cmake_minimum_required(VERSION 3.16)
project(GEvents LANGUAGES CXX)

add_library(gevents src/HashTable.cpp src/Text.cpp)
target_include_directories(gevents PUBLIC include)
target_compile_features(gevents PUBLIC cxx_std_17)
set_target_properties(gevents PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(gevents PRIVATE -Wall -Wextra -Wpedantic -Werror)
elseif(MSVC)
    target_compile_options(gevents PRIVATE /W4 /WX)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(hash_table_tests tests/hash_table_tests.cpp)
    target_link_libraries(hash_table_tests PRIVATE gevents)
    add_test(NAME hash_table COMMAND hash_table_tests)
endif()
